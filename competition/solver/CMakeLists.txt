cmake_minimum_required(VERSION 3.16)
project(contest_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(REPLAY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/replay")
add_library(solver_core core.cpp candidates.cpp supply.cpp solver.cpp
    "${REPLAY_DIR}/src/types.cpp" "${REPLAY_DIR}/src/transition.cpp" "${REPLAY_DIR}/src/replay.cpp")
target_include_directories(solver_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${REPLAY_DIR}/include")
add_executable(solver main.cpp)
target_link_libraries(solver PRIVATE solver_core)
add_executable(contest_solver contest/main.cpp contest/adapter.cpp)
target_link_libraries(contest_solver PRIVATE solver_core)
if(MSVC)
    target_compile_options(solver_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(solver_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
